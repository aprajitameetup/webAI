# Web AI Stack — Phase 0 + Slice 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the `webai` pnpm monorepo, port the prototype learning hub into `apps/hub` (Phase 0), then build `@webai/core` + `@webai/react` and a flagship on-device LLM chat demo end-to-end (Slice 1).

**Architecture:** pnpm workspaces. `@webai/core` is a framework-agnostic engine (detect → route → fallback) exposing an async-iterator token stream; `@webai/react` wraps it in hooks/components; `apps/hub` consumes both and showcases live demos. On-device inference uses WebLLM over WebGPU in a Web Worker with OPFS-cached weights; the fallback path streams from a server endpoint.

**Tech Stack:** pnpm · Vite · React 18 · TypeScript · Vitest · WebLLM (@mlc-ai/web-llm) · Vercel/Netlify (COOP/COEP).

**Verification note:** Browser-platform code (WebGPU, WebLLM, OPFS, Web Audio, Workers) cannot run in Node/Vitest. Those tasks are verified **manually in the browser** with an explicit checklist; pure logic (capability shape, router selection, stream adapters, hook state) is unit-tested in Vitest with mocks.

```ts
export interface Capabilities {
  webgpu: boolean; webnn: boolean; wasm: boolean; workers: boolean;
  opfs: boolean; webtransport: boolean; webrtc: boolean; sse: boolean;
  webcodecs: boolean; webaudio: boolean; builtinAI: boolean;
  crossOriginIsolated: boolean;
}

const safe = (fn: () => boolean): boolean => { try { return !!fn(); } catch { return false; } };

export function detectCapabilities(): Capabilities {
  return {
    webgpu: safe(() => "gpu" in navigator),
    webnn: safe(() => "ml" in navigator),
    wasm: safe(() => typeof WebAssembly === "object"),
    workers: safe(() => typeof Worker !== "undefined"),
    opfs: safe(() => !!navigator.storage && "getDirectory" in navigator.storage),
    webtransport: safe(() => typeof (globalThis as any).WebTransport !== "undefined"),
    webrtc: safe(() => typeof RTCPeerConnection !== "undefined"),
    sse: safe(() => typeof EventSource !== "undefined"),
    webcodecs: safe(() => typeof (globalThis as any).VideoEncoder !== "undefined"),
    webaudio: safe(() => typeof (window.AudioContext || (window as any).webkitAudioContext) !== "undefined"),
    builtinAI: safe(() => "LanguageModel" in self || !!(window as any).ai),
    crossOriginIsolated: safe(() => self.crossOriginIsolated === true),
  };
}
```

```ts
import type { Engine, Message, Token } from "../types";

export function createServerEngine(opts: { serverUrl: string }): Engine {
  return {
    source: "server",
    async *chat(messages: Message[]): AsyncIterable<Token> {
      const res = await fetch(opts.serverUrl, {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ messages }),
      });
      if (!res.ok || !res.body) throw new Error(`server ${(res as any).status ?? ""}`.trim());
      const reader = res.body.getReader();
      const dec = new TextDecoder();
      let buf = "";
      for (;;) {
        const { done, value } = await reader.read();
        if (done) break;
        buf += dec.decode(value, { stream: true });
        const lines = buf.split("\n");
        buf = lines.pop() ?? "";
        for (const line of lines) {
          if (!line.startsWith("data: ")) continue;
          const data = line.slice(6).trim();
          if (data === "[DONE]") { yield { delta: "", done: true }; return; }
          try { yield { delta: JSON.parse(data).delta ?? "", done: false }; } catch { /* skip */ }
        }
      }
      yield { delta: "", done: true };
    },
  };
}
```

```ts
import { CreateWebWorkerMLCEngine, type MLCEngineInterface } from "@mlc-ai/web-llm";
import type { Engine, Message, Token } from "../types";

const DEFAULT_MODEL = "Llama-3.2-1B-Instruct-q4f32_1-MLC";

export async function createLocalEngine(opts: {
  model?: string; onProgress?: (p: number) => void;
}): Promise<Engine> {
  const worker = new Worker(new URL("./webllm.worker.ts", import.meta.url), { type: "module" });
  const engine: MLCEngineInterface = await CreateWebWorkerMLCEngine(
    worker,
    opts.model ?? DEFAULT_MODEL,
    { initProgressCallback: (r) => opts.onProgress?.(r.progress) }
  );
  return {
    source: "on-device",
    async *chat(messages: Message[]): AsyncIterable<Token> {
      const stream = await engine.chat.completions.create({ messages, stream: true });
      for await (const chunk of stream) {
        const delta = chunk.choices[0]?.delta?.content ?? "";
        if (delta) yield { delta, done: false };
      }
      yield { delta: "", done: true };
    },
  };
}
```

```ts
import type { Capabilities, CreateEngineOptions, Engine, EngineSource } from "./types";
import { detectCapabilities } from "./detect";
import { createServerEngine } from "./engines/server";
import { createLocalEngine } from "./engines/webllm";

export function selectSource(caps: Capabilities, opts: CreateEngineOptions): EngineSource {
  if (opts.prefer) return opts.prefer;
  if (caps.webgpu) return "on-device";
  return "server";
}

export async function createEngine(opts: CreateEngineOptions = {}): Promise<Engine> {
  const caps = detectCapabilities();
  const source = selectSource(caps, opts);
  if (source === "server") {
    if (!opts.serverUrl) throw new Error("serverUrl required for server fallback");
    return createServerEngine({ serverUrl: opts.serverUrl });
  }
  return createLocalEngine({ model: opts.model, onProgress: opts.onProgress });
}
```

```ts
import { useCallback, useEffect, useRef, useState } from "react";
import { createEngine, type Engine, type Message, type CreateEngineOptions, type EngineSource } from "@webai/core";

export type LLMStatus = "detecting" | "downloading" | "ready" | "streaming" | "error";

export function useLocalLLM(opts: CreateEngineOptions = {}) {
  const [status, setStatus] = useState<LLMStatus>("detecting");
  const [progress, setProgress] = useState(0);
  const [messages, setMessages] = useState<Message[]>([]);
  const [source, setSource] = useState<EngineSource | null>(null);
  const [error, setError] = useState<string | null>(null);
  const engineRef = useRef<Engine | null>(null);

  useEffect(() => {
    let alive = true;
    setStatus("detecting");
    createEngine({ ...opts, onProgress: (p) => { if (alive) { setStatus("downloading"); setProgress(p); } } })
      .then((engine) => { if (!alive) return; engineRef.current = engine; setSource(engine.source); setStatus("ready"); })
      .catch((e) => { if (alive) { setError(String(e?.message ?? e)); setStatus("error"); } });
    return () => { alive = false; };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const send = useCallback(async (text: string) => {
    const engine = engineRef.current;
    if (!engine) return;
    const userMsg: Message = { role: "user", content: text };
    const history = [...messages, userMsg];
    setMessages([...history, { role: "assistant", content: "" }]);
    setStatus("streaming");
    try {
      let acc = "";
      for await (const tok of engine.chat(history)) {
        if (tok.done) break;
        acc += tok.delta;
        setMessages((prev) => { const next = [...prev]; next[next.length - 1] = { role: "assistant", content: acc }; return next; });
      }
      setStatus("ready");
    } catch (e) { setError(String((e as Error)?.message ?? e)); setStatus("error"); }
  }, [messages]);

  return { status, progress, messages, source, error, send };
}
```
